technical-showcase/sql/oracle_incremental_staging_cte.sql 
WITH RankedTelemetry AS (
  SELECT t.transaction_id, t.customer_account_id, TRUNC(t.transaction_timestamp) AS date_key, t.net_amount_eur,
         ROW_NUMBER() OVER (PARTITION BY t.transaction_id ORDER BY t.revision_sequence DESC) AS row_rank
  FROM erp_owner.sales_transactions_partitioned t
  WHERE t.transaction_timestamp >= TO_TIMESTAMP(:RangeStart, 'YYYY-MM-DD HH24:MI:SS')
    AND t.transaction_timestamp <  TO_TIMESTAMP(:RangeEnd, 'YYYY-MM-DD HH24:MI:SS')
    AND t.status_code IN ('SETTLED', 'POSTED', 'AUDITED')
)
SELECT transaction_id, customer_account_id, date_key, net_amount_eur
FROM RankedTelemetry WHERE row_rank = 1;
