technical-showcase/dax/calculation_groups_and_semi_additive.dax 
-- Dim_Date joins to Fact on integer DateKey (YYYYMMDD) for VertiPaq compression,
-- while Dim_Date[Date] (DATE type, marked as Date Table) drives Time Intelligence.
CALCULATION GROUP 'Time Intelligence'[Period]

  CALCULATION ITEM "YTD" =
    CALCULATE(
        SELECTEDMEASURE(),
        DATESYTD('Dim_Date'[Date])
    )

  CALCULATION ITEM "YoY Growth %" =
    VAR _Current = SELECTEDMEASURE()
    VAR _Prior   = CALCULATE(SELECTEDMEASURE(), SAMEPERIODLASTYEAR('Dim_Date'[Date]))
    RETURN DIVIDE(_Current - _Prior, _Prior, BLANK())

-- Semi-Additive Closing Balance (Single Storage Engine scan; avoids FILTER(ALL(...)))
[Closing Inventory Value] =
VAR _LastSnapshotDateKey =
    CALCULATE(MAX('Fact_Inventory_Snapshot'[SnapshotDateKey]))
RETURN
    CALCULATE(
        SUM('Fact_Inventory_Snapshot'[EndingValuation_EUR]),
        KEEPFILTERS('Dim_Date'[DateKey] = _LastSnapshotDateKey)
    )

technical-showcase/sql/oracle_incremental_staging_cte.sql 
WITH RankedTelemetry AS (
    SELECT
        t.transaction_id, t.customer_account_id, t.store_id,
        TRUNC(t.transaction_timestamp) AS date_key,
        t.net_amount_eur, t.tax_amount_eur,
        ROW_NUMBER() OVER (
            PARTITION BY t.transaction_id
            ORDER BY t.revision_sequence DESC, t.transaction_timestamp DESC
        ) AS row_rank
    FROM erp_owner.sales_transactions_partitioned t
    WHERE
        -- Sargable bounds map directly to Power BI RangeStart / RangeEnd for partition pruning
        t.transaction_timestamp >= TO_TIMESTAMP(:RangeStart, 'YYYY-MM-DD HH24:MI:SS')
        AND t.transaction_timestamp <  TO_TIMESTAMP(:RangeEnd, 'YYYY-MM-DD HH24:MI:SS')
        AND t.status_code IN ('SETTLED', 'POSTED', 'AUDITED')
)
SELECT transaction_id, customer_account_id, date_key, net_amount_eur,
       DENSE_RANK() OVER (PARTITION BY date_key ORDER BY net_amount_eur DESC) AS daily_rank
FROM RankedTelemetry
WHERE row_rank = 1;

spark = (
    SparkSession.builder
    .appName("Fabric_Gold_Sales_DirectLake_Optimizer")
    .config("spark.sql.parquet.vorder.enabled", "true")
    .config("spark.microsoft.delta.optimizeWrite.enabled", "true")
    .config("spark.microsoft.delta.autoCompact.enabled", "true")
    .getOrCreate()
)

df_gold_fact = (
    spark.read.format("delta").load(silver_path)
    .filter(F.col("status_code") == "SETTLED")
    .withColumn("DateKey", F.date_format("transaction_timestamp", "yyyyMMdd").cast("int"))
    .withColumn("TimeKey", F.date_format("transaction_timestamp", "HHmmss").cast("int"))
    .select("transaction_id", "DateKey", "TimeKey", "customer_account_id", "net_amount_eur")
)
df_gold_fact.write.format("delta").mode("overwrite").saveAsTable("gold_fact_sales_telemetry")
spark.sql("OPTIMIZE gold_fact_sales_telemetry ZORDER BY (DateKey, customer_account_id)")
